# P79 GLM sidecar cron — opt-in auto-trigger with ntfy failure alerts
#
# To install: crontab scripts/maintenance/crontab.txt
# To uninstall: crontab -r
# To preview without installing: cat scripts/maintenance/crontab.txt
# To check installed: crontab -l
#
# Each entry wrapped with notify_on_fail.sh — sends ntfy.sh alert (priority=high)
# on non-zero exit code with last 500 chars of output.
# Topic: $NTFY_TOPIC env or default p79-exp-dgx-spark.

REPO=/home/jiaming/workspace/Cost-Aware-Routing-for-Web-Usage-Agents
SHELL=/bin/bash
NTFY_TOPIC=p79-exp-dgx-spark

# every 10 min: sync _status/cells/*.md frontmatter from condition_summary_v2.json
# (skips active+pid cells iff PID alive; auto-clears stale dead PIDs)
*/10 * * * * cd $REPO && bash scripts/maintenance/notify_on_fail.sh "glm-update-cells" -- make glm-update-cells APPLY=1 >> logs/cron/glm_update_cells.log 2>&1

# every 5 min: scan recent runner / watchdog / cron logs for ERROR / Traceback / OOM /
# NOT_LOGGED_IN / Timeout. Output JSON consumed by §2.5 active-errors subsection.
*/5 * * * * cd $REPO && bash scripts/maintenance/notify_on_fail.sh "error-scan" -- make error-scan >> logs/cron/error_scan.log 2>&1

# PLAYBOOK §1+§2 refresh: REMOVED 2026-05-13. Replaced by manual hermes agent

# every Sunday 00:00: dead link scan, output report to logs (don't notify on this — it always exits 1 if links found)
0 0 * * 0 cd $REPO && make check-links > logs/cron/dead_links_$(date +\%Y\%m\%d).log 2>&1

# every 5 min: poll Myriad qstat via SSH chain (DGX → quark Tailscale → myriad);
# diff against logs/cron/myriad_state.json and push ntfy on transitions
# (qw→r, r→GONE/finished). On GONE includes tail of .err for context.
# SSH chain failures exit 0 silently (transient blip, watchdog handles persistent).
*/5 * * * * cd $REPO && bash scripts/maintenance/notify_on_fail.sh "myriad-watcher" -- python3 scripts/maintenance/glm/myriad_watcher.py >> logs/cron/myriad_watcher.log 2>&1

# every 15 min: rsync A100 Phase 1a results → DGX local + trigger incremental analysis.
# WHY: Phase 1a fire runs on condense-a100 (results in /mnt/scratch/), but P79 GLM
# Phase 2 sidecars (glm-update-cells, error-scan) read DGX-local results/. Without
# bridge, 18-day fire = DGX monitoring stack stale entire run. See M1 in
# docs/checkpoints/process/cross_system_docker_audit_3way_diff_2026-05-15.md.
*/15 * * * * cd $REPO && bash scripts/maintenance/notify_on_fail.sh "sync-a100" -- bash scripts/maintenance/sync_a100_results.sh >> logs/cron/sync_a100_results.log 2>&1

# twice daily (08:30/20:30, was 6h — travel noise-cut 2026-07-07): paper-grade integrity check of the A100 Phase-1a
# fire — SSH A100, validate completed conditions (episodes==scored [B-1834] / parse_err
# + benchmark_noise==0 / som·vision IMAGE PRESENCE [B-1828→1832→1835 image-loss
# regression class] / B0 cost coverage) + in-progress run health (images landing, no
# UnboundLocalError/deferred-save-failed flood) + manifest ghost/over-complete. ntfy
# verdict ✅/🔴 each run; full output → logs/cron/paper_grade_check.log. (set 2026-05-22)
30 8,20 * * * cd $REPO && bash scripts/maintenance/notify_on_fail.sh "paper-grade-check" -- bash scripts/maintenance/paper_grade_check.sh >> logs/cron/paper_grade_check.log 2>&1
