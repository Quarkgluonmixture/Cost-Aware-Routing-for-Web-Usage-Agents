% P79 FIX, two defects in the template's header block.
%
% (1) The template writes the section mark and the chapter mark to the left and
%     right of the SAME header line. With titles as long as this dissertation's
%     the two overprint each other, giving a 104pt overfull box and two
%     unreadable titles stacked on top of one another on most pages. The
%     standard two-sided arrangement is used instead: chapter on the verso,
%     section on the recto, one mark per page. The page number is in the footer
%     either way, so no information is lost.
%
% (2) The template's derived styles are written as \fancypagestyle{plain}{...},
%     which under fancyhdr v4 inherits from the empty prototype and then leaks
%     its \fancyhead[R] assignment into the live style. After the first chapter
%     opening, every page in the document carried the chapter title on the right
%     as well, for the rest of the run. Naming the main layout and inheriting
%     from it with the [base] argument re-establishes all four corners each time
%     a style is selected, so nothing leaks.
\fancypagestyle{main}{%
  \fancyhf{}%
  \fancyhead[LE]{\nouppercase{\leftmark}}%                                               Chapter title on even (left-hand) pages.
  \fancyhead[RO]{\nouppercase{\rightmark}}%                                              Section title on odd (right-hand) pages.
  \fancyfoot[LE,RO]{\thepage\ of\ \pageref{LastPage}}%                                   Page number counter.
  \renewcommand{\headrulewidth}{0.4pt}%                                                  Header underline thickness.
  \renewcommand{\footrulewidth}{0.4pt}%                                                  Footer overline thickness.
}
