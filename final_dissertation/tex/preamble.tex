% =============================================================================
%  P79 project macros.
%
%  Everything the UCL template supplies lives in main.tex, in the template's own
%  order, so that a template update can be diffed. This file holds only what
%  this dissertation adds on top: the term lock, the scope box, and the chapter
%  hand-off. Nothing here changes page geometry, fonts or bibliography.
% =============================================================================

% ---------- caption and list style -------------------------------------------
% The template leaves captions at document default. These figures carry long
% captions by design (the argument was moved out of the images and into the
% captions), so captions are set one size down and the label is bolded to keep
% the caption visually distinct from body text.
\captionsetup{font=small,labelfont=bf}
\captionsetup[table]{skip=4pt}
\usepackage{enumitem}
\setlist{itemsep=2pt,topsep=3pt,parsep=0pt}
\usepackage{microtype}
\usepackage{url}

% The template's 40mm binding margins leave a 13cm text block, which is narrow
% for a document whose prose is full of underscored identifiers such as
% \texttt{has\_reference\_image}. Without these two lines such a word cannot be
% broken and pushes a line into the margin.
\usepackage[htt]{hyphenat}
\emergencystretch=1.5em

% Let biblatex break long URLs and arXiv identifiers rather than run them into
% the margin. Without these the bibliography contributes the widest overfull box
% in the document.
\setcounter{biburllcpenalty}{7000}
\setcounter{biburlucpenalty}{8000}
\setcounter{biburlnumpenalty}{7000}

% ---------- semantic macros --------------------------------------------------
% Term lock (final_dissertation/TERMS.md): one concept, one name, one spelling.
\newcommand{\pp}{\,pp}                       % percentage points

\newcommand{\mVis}{\textsc{vision}}
\newcommand{\mPT}{\textsc{P-text}}
\newcommand{\mPP}{\textsc{P-prompt}}
\newcommand{\mPS}{\textsc{P-SoM}}
\newcommand{\sommarks}{\texttt{[SOM\_MARKS]}}
\newcommand{\cell}[1]{\textsf{#1}}

% Boxed "scope caveat" used wherever a headline number is stated. The house rule
% for this dissertation is that no ceiling number appears without its comparator
% and its cell set attached; the box makes violations visually obvious in draft.
\usepackage[framemethod=TikZ]{mdframed}
\newmdenv[linewidth=0.6pt,linecolor=black!35,backgroundcolor=black!3,
          innertopmargin=5pt,innerbottommargin=5pt,
          innerleftmargin=8pt,innerrightmargin=8pt,
          skipabove=8pt,skipbelow=8pt,roundcorner=2pt]{scopebox}

\newcommand{\chapterhandoff}[1]{%
  \bigskip\noindent\begin{scopebox}\textit{#1}\end{scopebox}}
