% =============================================================================
%  P79 project macros.
%
%  Everything the UCL template supplies lives in main.tex, in the template's own
%  order, so that a template update can be diffed. This file holds only what
%  this dissertation adds on top: the term lock, the scope box, and the chapter
%  hand-off. Nothing here changes page geometry, fonts or bibliography.
% =============================================================================

% ---------- caption and list style -------------------------------------------
% The template leaves captions at document default. These figures carry long
% captions by design (the argument was moved out of the images and into the
% captions), so captions are set one size down and the label is bolded to keep
% the caption visually distinct from body text.
\captionsetup{font=small,labelfont=bf}
\captionsetup[table]{skip=4pt}
\usepackage{enumitem}
\setlist{itemsep=2pt,topsep=3pt,parsep=0pt}
\usepackage{microtype}

% Command-line flags must survive the typesetter. lmodern's T1 typewriter face
% carries the `--' ligature that the original Computer Modern tt face does not,
% so \texttt{--allow-partial} was reaching the PDF as an EN DASH: the reader who
% copies it out of a reproducibility appendix gets a flag no shell accepts.
% Verified before the fix by extracting the PDF text and reading the bytes:
% `e2 80 93' (U+2013) where a double hyphen was written. microtype can switch
% the ligature off for one family, which is narrower than swapping the font and
% cannot leak into body text.
\DisableLigatures{encoding = T1, family = tt*}
\usepackage{url}

% ---------- float placement ---------------------------------------------------
% Every figure/table in the chapters is `[t]`-only. That is fine for an
% isolated float, but a chapter carrying several of them (ch4 has twelve) lets
% LaTeX's float queue back up across many pages with nothing to force a
% discharge, so a float can drift past the prose that discusses it and even
% past the chapter's own closing summary. `placeins` adds `\FloatBarrier`,
% inserted at each section boundary in the chapters that carry floats, which
% forces the queue to clear there rather than several sections later.
\usepackage{placeins}
% `float` supplies [H], used by the appendix tables. Those tables are not
% illustrations that may drift to a convenient page: each one belongs to the
% section that introduces it, and when LaTeX deferred them the reader met
% Table A.1 under heading A.2, Table C.1 splitting a six-step protocol between
% steps 3 and 4, and Table D.2 under D.5. [H] pins them where they are written.
\usepackage{float}
\renewcommand{\topfraction}{0.9}
\renewcommand{\floatpagefraction}{0.75}
\setcounter{topnumber}{4}
\setcounter{totalnumber}{6}

% The template's 40mm binding margins leave a 13cm text block, which is narrow
% for a document whose prose is full of underscored identifiers such as
% \texttt{has\_reference\_image}. Without these two lines such a word cannot be
% broken and pushes a line into the margin.
\usepackage[htt]{hyphenat}
\emergencystretch=1.5em

% File paths and artefact names. The [htt] option above lets LaTeX hyphenate
% inside \texttt, which is what keeps identifiers like has\_reference\_image
% out of the margin --- but applied to a PATH it inserts a line-end hyphen that
% reads as part of the filename. In a reproducibility manifest that is worse
% than an overfull line: `router\_objective\_-ordering.md` is not a file anyone
% can open. This macro keeps the typewriter face, forbids the hyphen outright
% (\hyphenchar = -1), and instead offers a break opportunity after each
% underscore, so long paths still wrap.
\newcommand{\artefact}[1]{{%
  \ttfamily\hyphenchar\font=-1\relax
  \renewcommand{\_}{\textunderscore\allowbreak}%
  #1}}

% Every one of this document's 121 \texttt uses is a code identifier, a config
% key or a file path, so the rule above is the right rule for all of them, not
% just for the artefact manifest: `NON\_PAPER\_-GRADE`, `phantom\_-text` and
% `max\_new\_-tokens` were all being rendered with a hyphen that is not in the
% identifier. Redefining \texttt applies the fix in one place instead of at 73
% call sites, and cannot drift away from \artefact.
\let\origtexttt\texttt
\renewcommand{\texttt}[1]{\artefact{#1}}

% Let biblatex break long URLs and arXiv identifiers rather than run them into
% the margin. Without these the bibliography contributes the widest overfull box
% in the document.
\setcounter{biburllcpenalty}{7000}
\setcounter{biburlucpenalty}{8000}
\setcounter{biburlnumpenalty}{7000}

% The Table of Acronyms is printed with the `long-booktabs' glossary style,
% whose first column header comes from \entryname --- and the glossaries default
% for that is "Notation". A page titled "Table of Acronyms" whose column then
% says "Notation" is naming the same thing two ways on one line.
% \AtBeginDocument, not here directly: the glossaries language module assigns
% \entryname at \begin{document}, so a plain \renewcommand in the preamble is
% overwritten before the glossary is ever printed --- which is exactly what the
% first attempt at this did, silently and with a clean compile.
\AtBeginDocument{%
  \renewcommand{\entryname}{Acronym}%
  \renewcommand{\descriptionname}{Description}%
}

% Boxed "scope caveat" used wherever a headline number is stated. The house rule
% for this dissertation is that no ceiling number appears without its comparator
% and its cell set attached; the box makes violations visually obvious in draft.

          innertopmargin=5pt,innerbottommargin=5pt,
          innerleftmargin=8pt,innerrightmargin=8pt,
          skipabove=8pt,skipbelow=8pt,roundcorner=2pt]{scopebox}

% The chapter hand-off uses its own non-breakable frame. When it shared the
% breakable `scopebox`, a hand-off that started near the foot of a page split
% across the break and left a single orphaned line -- "describe. Chapter 7
% takes up all three, and closes the dissertation." was alone on its own page.
% These boxes are never more than a few lines, so moving one whole is always
% cheaper than splitting it.
\newmdenv[linewidth=0.6pt,linecolor=black!35,backgroundcolor=black!3,
          innertopmargin=5pt,innerbottommargin=5pt,
          innerleftmargin=8pt,innerrightmargin=8pt,
          skipabove=8pt,skipbelow=8pt,roundcorner=2pt,
          nobreak=true]{handoffbox}

\newcommand{\chapterhandoff}[1]{%
  \bigskip\noindent\begin{handoffbox}\textit{#1}\end{handoffbox}}
